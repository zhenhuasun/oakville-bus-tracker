let ttIndex = null;
function buildTTIndex() {
  if (ttIndex || !TT || !TT.trips) return ttIndex;
  const byStop = new Map();
  TT.trips.forEach((tr, ti) => {
    tr[3].forEach((row, si) => {
      if (!byStop.has(row[0])) byStop.set(row[0], []);
      byStop.get(row[0]).push({ ti, si, dep: row[2], arr: row[1] });
    });
  });
  for (const arr of byStop.values()) arr.sort((a, b) => a.dep - b.dep);
  ttIndex = { byStop, neighbours: new Map() };
  return ttIndex;
}
function stopNameOf(id) { const s = stopById(id); return s ? s.name : id; }
function stopCodeOf(id) { const s = stopById(id); return s ? s.code : ""; }
function tripTerminalName(tr) {
  const last = tr[3][tr[3].length - 1];
  return stopNameOf(last[0]);
}
function neighboursWithin(stopId, maxM) {
  const idx = buildTTIndex();
  if (idx.neighbours.has(stopId)) return idx.neighbours.get(stopId);
  const s = stopById(stopId);
  const out = [];
  if (s && STOPS) {
    for (const o of STOPS.stops) {
      if (o.id === stopId) continue;
      const d = Math.round(haversineM(s.lat, s.lon, o.lat, o.lon));
      if (d <= maxM) out.push({ id: o.id, dist: d });
    }
  }
  out.sort((a, b) => a.dist - b.dist);
  idx.neighbours.set(stopId, out);
  return out;
}
function currentOriginStopId() {

function currentDestStopId() {
  if (journeyState) return journeyState.d;
  const pair = currentPair();
  if (pair && pair.destination_stop) return pair.destination_stop.id;
  const d = DATA.destinations.find(x => x.id === dest);
  return d && d.candidate_stops && d.candidate_stops[0] ? d.candidate_stops[0].id : null;
}
function makeLeg(tr, ti, boardEntry, alightIdx) {
  const row = tr[3][alightIdx];
  return { route: tr[0], service: tr[1], original: tr[2], tripIdx: ti,
           boardStop: tr[3][boardEntry.si][0], boardDep: boardEntry.dep,
           alightStop: row[0], alightArr: row[1],
           stops: alightIdx - boardEntry.si };
}
function legLive(leg, dayOffset) {
  // Real-time applies only to today's legs; future-day legs are timetable.
  if (dayOffset > 0) return { live: false, minutesAway: null };
  const tu = RT.trips.get(leg.original + "|" + leg.route);
  if (!tu) return { live: false, minutesAway: null };
  const stu = tu.stops.find(s => s.stop_id === leg.boardStop);
  if (stu && stu.time) {
    return { live: true, minutesAway: Math.round((stu.time * 1000 - Date.now()) / 60000) };
  }
  return { live: false, minutesAway: null };
}

// Full journey search: direct first, then at most one transfer.
// Returns {options, dayOffset, dayDate} for the first day (today, then the
// next 6 days) that has any option, or null. Options are sorted by fewest
// transfers, then earliest arrival, then least walking; at most 3 total.
function searchJourneys(originStop, destStop, now) {
  if (!TT || !TT.trips || !DATA || !originStop || !destStop || originStop === destStop) return null;
  buildTTIndex();
  const idx = ttIndex;
  const base = torontoParts(now);
  for (let d = 0; d < 7; d++) {
    const day = torontoParts(new Date(now.getTime() + d * 86400000));
    const svc = activeServices(DATA.meta, day.dateStr, day.weekday);
    const threshold = d === 0 ? base.minutes - 1 : 0;
    const fromList = (idx.byStop.get(originStop) || []).filter(e => svc.has(TT.trips[e.ti][1]));
    const options = [];
    // (a) Direct rides: earliest departures first, up to 3.
    for (const e of fromList) {
      if (e.dep < threshold) continue;
      const tr = TT.trips[e.ti];
      let ai = -1;
      for (let si = e.si + 1; si < tr[3].length; si++) if (tr[3][si][0] === destStop) { ai = si; break; }
      if (ai < 0) continue;
      options.push({ transfers: 0, legs: [makeLeg(tr, e.ti, e, ai)], walkM: 0, walkMin: 0 });
      if (options.length >= 3) break;
    }
    // (b) One-transfer rides.
    const transfers = [];
    const seen = new Set();
    const addTransfer = (tr1, e1, S, S2, walkM, walkMin, arrS) => {
      const gap = Math.max(MIN_TRANSFER_MIN, walkMin);
      const list = (idx.byStop.get(S2) || []).filter(e => svc.has(TT.trips[e.ti][1]));
      let taken = 0;
      for (const e2 of list) {
        if (e2.dep < arrS + gap) continue;
        if (e2.ti === e1.ti) continue; // same trip = a direct ride, not a transfer
        const tr2 = TT.trips[e2.ti];
        let ai = -1;
        for (let si = e2.si + 1; si < tr2[3].length; si++) if (tr2[3][si][0] === destStop) { ai = si; break; }
        if (ai < 0) continue;
        const leg1 = makeLeg(tr1, e1.ti, e1, tr1[3].findIndex((r, si) => si > e1.si && r[0] === S));
        const key = tr1[0] + "|" + tr2[0] + "|" + S + "|" + S2 + "|" + e1.dep;
        if (seen.has(key)) continue;
        seen.add(key);
        transfers.push({ transfers: 1, legs: [leg1, makeLeg(tr2, e2.ti, e2, ai)],
                        walkM, walkMin, transferStop: S, transferBoardStop: S2 });
        if (++taken >= 2) break;
      }
    };
    let t1count = 0;
    for (const e1 of fromList) {
      if (e1.dep < threshold) continue;
      if (t1count++ >= 40) break; // first 40 departures from the origin: enough for a day's options
      const tr1 = TT.trips[e1.ti];
      for (let si = e1.si + 1; si < tr1[3].length; si++) {
        const S = tr1[3][si][0], arrS = tr1[3][si][1];
        if (S === originStop || S === destStop) continue;
        addTransfer(tr1, e1, S, S, 0, 0, arrS); // same stop: no walking
        for (const nb of neighboursWithin(S, MAX_WALK_M)) {
          if (nb.id === destStop) continue; // walking to the destination itself is not a bus transfer
          addTransfer(tr1, e1, S, nb.id, nb.dist, Math.ceil(nb.dist / WALK_SPEED_M_PER_MIN), arrS);
        }
      }
    }
    transfers.sort((a, b) => a.legs[1].alightArr - b.legs[1].alightArr || a.walkM - b.walkM);
    const merged = options.concat(transfers);
    merged.sort((a, b) => a.transfers - b.transfers
      || a.legs[a.legs.length - 1].alightArr - b.legs[b.legs.length - 1].alightArr
      || a.walkM - b.walkM);
    if (merged.length > 0) {
      return { options: merged.slice(0, 3), dayOffset: d, dayDate: day.display };
    }
  }
  return null;
}
function placeStopIds(p) {
  // A place is reachable at its nearest stop, but also at other stops
  // within 300 m of the place: a trip may serve the second-nearest stop
  // in a direction the nearest stop does not serve. Trying those stops
  // (nearest first) keeps a real ride from being missed.
  const ids = [p.stop.id];
  if (p.lat && p.lon && STOPS) {
    const near = STOPS.stops
      .map(s => ({ id: s.id, dist: Math.round(haversineM(p.lat, p.lon, s.lat, s.lon)) }))
      .filter(x => x.dist <= 300 && x.id !== p.stop.id)
      .sort((a, b) => a.dist - b.dist).slice(0, 3);
    for (const n of near) ids.push(n.id);
  }
  return ids;
}
function savedPlaceStopIds(p) {
  if (p.lat && p.lon) return placeStopIds({ stop: { id: p.stopId }, lat: p.lat, lon: p.lon });
  return [p.stopId];
}
function setJourney(oStop, dStop, oLabel, dLabel) {
  if (!TT) { setSavedNote(t("cantStart")); return false; }
  // dStop may be a single stop id, or a list of candidate stops for one
  // place (prototype destination candidates, or stops near a place).
  // The best journey across all candidates wins, by day, then fewest
  // transfers, then earliest arrival. journeyState.d is the stop that won.
  const destIds = (Array.isArray(dStop) ? dStop : [dStop]).filter(ds => ds && ds !== oStop);
  if (!oStop || destIds.length === 0) { setSavedNote(t("noJourney")); render(); return false; }
  let best = null;
  for (const ds of destIds) {
    const res = searchJourneys(oStop, ds, new Date());
    if (!res || !res.options.length) continue;
    if (!best) { best = { res, ds }; continue; }
    const a = best.res, b = res;
    const aArr = a.options[0].legs[a.options[0].legs.length - 1].alightArr;
    const bArr = b.options[0].legs[b.options[0].legs.length - 1].alightArr;
    if (b.dayOffset < a.dayOffset ||
        (b.dayOffset === a.dayOffset && (b.options[0].transfers < a.options[0].transfers ||
          (b.options[0].transfers === a.options[0].transfers && bArr < aArr)))) {
      best = { res, ds };
    }
  }
  journeyState = { o: oStop, d: best ? best.ds : destIds[0],
    labelO: oLabel || stopNameOf(oStop), labelD: dLabel || stopNameOf(best ? best.ds : destIds[0]),
    options: best ? best.res.options : null,
    dayOffset: best ? best.res.dayOffset : 0, dayDate: best ? best.res.dayDate : "" };
  render();
  return true;
}
function swapJourney() {
  if (!journeyState) return;
  setJourney(journeyState.d, journeyState.o, journeyState.labelD, journeyState.labelO);
}
function journeyOptionSummary(opt) {
  const first = opt.legs[0], last = opt.legs[opt.legs.length - 1];
  const total = Math.round(last.alightArr - first.boardDep);
  let s = fmtTime(first.boardDep) + " → " + fmtTime(last.alightArr) + " · " +
    t("totalJourney") + " " + total + " " + t("min") + " · " +
    (opt.transfers === 0 ? t("direct") : t("oneTransfer"));
  if (opt.transfers === 1) {
    s += " (" + t("transferAt") + " " + stopNameOf(opt.transferStop) +
      (opt.walkM > 0 ? " · " + t("walk") + " " + opt.walkM + " m · " + opt.walkMin + " " + t("walkMin") : " · " + t("sameStop")) + ")";
  }
  return s;
}

  const box = document.getElementById("alerts"); box.innerHTML = "";

  for (const a of RT.alerts) {
    if (a.route_ids.length > 0 && !a.route_ids.some(r => routes.includes(r))) continue;
